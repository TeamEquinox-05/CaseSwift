```python
# ChatBot/conversation_store.py
import json
import os
from datetime import datetime

def save_conversation(case_id, session_id, data):
    dir_path = f"chat_sessions/{session_id}"
    os.makedirs(dir_path, exist_ok=True)
    
    with open(f"{dir_path}/conversation.json", "w") as f:
        json.dump(data, f, indent=2)
    
def load_conversation(session_id):
    path = f"chat_sessions/{session_id}/conversation.json"
    if os.path.exists(path):
        with open(path, "r") as f:
            return json.load(f)
    return None

def save_structured_case(case_id, structured_data):
    dir_path = f"cases/{case_id}"
    os.makedirs(dir_path, exist_ok=True)
    
    filename = f"{case_id}_{datetime.now().strftime('%Y%m%d_%H%M%S')}.json"
    with open(f"{dir_path}/{filename}", "w") as f:
        json.dump(structured_data, f, indent=2)
```

4. **Add Endpoints to server.js:**
```javascript
const ConversationalCase = require('./models/ConversationalCase');

// Save conversation progress
app.post('/api/conversation/save', async (req, res) => {
  const { caseId, sessionId, conversationHistory, extractedData, progress } = req.body;
  
  const conversationalCase = await ConversationalCase.findOneAndUpdate(
    { sessionId },
    {
      caseId,
      sessionId,
      conversationHistory,
      extractedData,
      progress,
      updatedAt: new Date()
    },
    { upsert: true, new: true }
  );
  
  res.json({ success: true, data: conversationalCase });
});

// Get conversation
app.get('/api/conversation/:sessionId', async (req, res) => {
  const conversation = await ConversationalCase.findOne({ 
    sessionId: req.params.sessionId 
  });
  res.json(conversation);
});

// Save final structured case
app.put('/api/case/:caseId/structured-data', async (req, res) => {
  const { structuredCaseData } = req.body;
  
  await ConversationalCase.findOneAndUpdate(
    { caseId: req.params.caseId },
    { 
      structuredCaseData, 
      isComplete: true,
      updatedAt: new Date()
    }
  );
  
  res.json({ success: true });
});
```
